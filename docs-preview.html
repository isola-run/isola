Quick Start

Create your first sandbox in under a minute

from isola import Isola

client = Isola(base_url="http://localhost:8080")

# Create a sandbox
sandbox = client.sandboxes.create(
    pod_template={
        "containers": [{"name": "main", "image": "python:3.12-slim"}]
    }
)

# Run a command and get the result
result = sandbox.commands.run("python", "-c", "print('Hello from the sandbox!')")
print(result.stdout)  # Hello from the sandbox!

# Stream output in real-time
cmd = sandbox.commands.spawn("pip", "install", "numpy")
for chunk in cmd.stdout:
    print(chunk, end="")
cmd.wait()
# Create a sandbox
curl -X POST http://localhost:8080/sandboxes \
  -H "Content-Type: application/json" \
  -d '{
    "podTemplate": {
      "containers": [{"name": "main", "image": "python:3.12-slim"}]
    }
  }'

# Run a command (returns 202 with commandId)
curl -X POST http://localhost:8080/sandboxes/{id}/commands \
  -H "Content-Type: application/json" \
  -d '{"args": ["python", "-c", "print(\"Hello!\")"]}'

# Stream stdout (Server-Sent Events)
curl -N http://localhost:8080/sandboxes/{id}/commands/{cmdId}/stdout
# Install the operator
helm install isola oci://ghcr.io/isola-ai/charts/isola \
  --namespace isola-system \
  --create-namespace

# Create a sandbox via CRD
kubectl apply -f - <<EOF
apiVersion: sandbox.isola.run/v1alpha1
kind: Sandbox
metadata:
  name: my-sandbox
  namespace: isola-sandboxes
spec:
  activeDeadlineSeconds: 3600
  podTemplate:
    containers:
      - name: main
        image: python:3.12-slim
EOF

Quick start - create a sandbox:

  1. Create a SandboxTemplate:

     cat <<EOF | kubectl apply -f -
     apiVersion: sandbox.isola.run/v1alpha1
     kind: SandboxTemplate
     metadata:
       name: python
       namespace: {{ .Values.sandboxNamespace }}
     spec:
       timeoutSeconds: 300
       podTemplate:
         spec:
           containers:
           - name: main
             image: python:3.11-slim
             command: ["sleep", "infinity"]
     EOF

  2. Create a Sandbox:

     cat <<EOF | kubectl apply -f -
     apiVersion: sandbox.isola.run/v1alpha1
     kind: Sandbox
     metadata:
       name: my-sandbox
       namespace: {{ .Values.sandboxNamespace }}
     spec:
       templateRef:
         name: python
     EOF

  3. Check sandbox status:

     kubectl get sandbox -n {{ .Values.sandboxNamespace }}
